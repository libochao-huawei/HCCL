function(add_modules)
    set(options OPTION_RESERVED)
    set(oneValueArgs MODE)
    set(multiValueArgs TARGETS SUB_LIBS)

    cmake_parse_arguments(ARGS "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

    foreach(target ${ARGS_TARGETS})
        if (TARGET ${target} AND TARGET ${ARGS_SUB_LIBS})
            message(STATUS "=== Debug add_modules: target_link_libraries(${target} ${ARGS_MODE} ${ARGS_SUB_LIBS})")
            target_link_libraries(${target} ${ARGS_MODE} ${ARGS_SUB_LIBS})
        else()
            message(STATUS "=== Debug add_modules: target_link_libraries ${target} is not init or ${ARGS_SUB_LIBS} is None")
        endif()
    endforeach()
endfunction()

function(add_infer_modules host_name)
    if (NOT TARGET ${host_name}_infer_obj)
        add_library(${host_name}_infer_obj OBJECT)
        target_compile_definitions(${host_name}_infer_obj
                                    PRIVATE OPS_UTILS_LOG_SUB_MOD_NAME="OP_PROTO")
        target_include_directories(${host_name}_infer_obj
                                    PRIVATE ${OP_PROTO_INCLUDE})
        if(UT_TEST_ALL OR PROTO_UT OR PASS_UT OR PLUGIN_UT OR ONNX_PLUGIN_UT)
            target_compile_definitions(${host_name}_infer_obj PRIVATE ASCEND_OPSPROTO_UT)
            target_compile_options(${host_name}_infer_obj PRIVATE
                                    ${COMPILE_OP_MODE}
                                    -Dgoogle=ascend_private
                                    -mcmodel=medium)
            target_link_libraries(${host_name}_infer_obj
                                    PRIVATE $<BUILD_INTERFACE:intf_llt_pub_asan_cxx14>
                                    $<$<BOOL:${BUILD_OPEN_PROJECT}>:$<BUILD_INTERFACE:alog_headers>>
                                    alog
                                    op_common)
        else()
            target_compile_options(${host_name}_infer_obj PRIVATE
                                    ${COMPILE_OP_MODE}
                                    -Dgoogle=ascend_private
                                    -DDISABLE_COMPILE_V1
                                    -fvisibility=hidden
                                    -fPIC
                                    -fno-common)
            target_link_libraries(${host_name}_infer_obj
                                    PRIVATE $<BUILD_INTERFACE:intf_pub_cxx14>
                                    $<$<BOOL:${BUILD_OPEN_PROJECT}>:$<BUILD_INTERFACE:alog_headers>>
                                    $<$<NOT:$<BOOL:${BUILD_OPEN_PROJECT}>>:$<BUILD_INTERFACE:slog_headers>>
                                    alog
                                    op_common)
        endif()
    endif()
endfunction()

function(check_open_sources op_type open_dir)
    foreach(OPEN_DIR ${OPEN_SOURCE_LISTS})
        file(GLOB CURRENT_DIRS LIST_DIRECTORIES true RELATIVE ${OPS_DIR}/${OPEN_DIR} ${OPS_DIR}/${OPEN_DIR}/*)
        foreach(dir ${CURRENT_DIRS})
            if (EXISTS ${OPS_DIR}/${OPEN_DIR}/${dir}/${op_type})
                message(STATUS "found open dirs: ${OPS_DIR}/${OPEN_DIR}/${dir}/${op_type}")
                set(${open_dir} "${OPS_DIR}/${OPEN_DIR}/${dir}/${op_type}" PARENT_SCOPE)
                break()
            endif()
        endforeach()
    endforeach()
endfunction()

function(mc2_add_modules_sources)
    set(options OPTION_RESERVED)
    set(oneValueArgs HOSTNAME MODE DIR)
    set(multiValueArgs ACLNNTYPE)
    cmake_parse_arguments(MODULE "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})

    file(GLOB OPINFER_SRCS ${MODULE_DIR}/*_infershape*.cc)
    add_infer_modules(${MODULE_HOSTNAME})
    target_sources(${MODULE_HOSTNAME}_infer_obj ${MODULE_MODE} ${OPINFER_SRCS})

    file(GLOB OPAPI_SRCS ${MODULE_DIR}/op_api/*.cpp)
    add_opapi_modules(${MODULE_HOSTNAME})
    target_sources(${MODULE_HOSTNAME}_opapi_obj ${MODULE_MODE} ${OPAPI_SRCS})
    target_include_directories(${MODULE_HOSTNAME}_opapi_obj PRIVATE ${MODULE_DIR}/op_api)

    file(GLOB_RECURSE OPTILING_SRCS ${MODULE_DIR}/op_tiling/*_tiling*.cc)
    add_tiling_modules(${MODULE_HOSTNAME})
    target_sources(${MODULE_HOSTNAME}_tiling_obj ${MODULE_MODE} ${OPTILING_SRCS})
    target_include_directories(${MODULE_HOSTNAME}_tiling_obj PRIVATE ${MODULE_DIR}/op_tiling)

    list(LENGTH MODULE_ACLNNTYPE len)
    math(EXPR len "${len} - 1")
    foreach(i RANGE ${len})
        list(GET MODULE_ACLNNTYPE ${i} aclnn_type)
        if (${aclnn_type} STREQUAL "aclnn" OR
            ${aclnn_type} STREQUAL "aclnn_inner" OR
            ${aclnn_type} STREQUAL "aclnn_exclude")
            file(GLOB OPDEF_SRCS ${MODULE_DIR}/*_def*.cpp)
            target_sources(${MODULE_HOSTNAME}_opdef_${aclnn_type}_obj INTERFACE ${OPDEF_SRCS})
        else()
            message(FATAL_ERROR "ACLNN TYPE UNSPPORTED, ONLY SUPPORT aclnn/aclnn_inner/aclnn_exclude")
        endif()
    endforeach()
endfunction()

endfunction()

function(add_graph_plugin)
    get_filename_component(full_path ${CMAKE_CURRENT_SOURCE_DIR}/.. ABSOLUTE)
    get_filename_component(op_dir ${full_path} NAME)
    get_filename_component(op_dir ${op_dir} NAME)

function(add_modules_sources)
    set(options OPTION_RESERVED)
    set(oneValueArgs HOSTNAME MODE DIR ENABLE_IN_OPP)
    set(multiValueArgs OPTYPE ACLNNTYPE)

    cmake_parse_arguments(MODULE "${options}" "${oneValueArgs}" "${multiValueArgs}" ${ARGN})
    string(REGEX REPLACE "/op_host(/.*)?$" "" op_dir "${MODULE_DIR}")
    get_filename_component(op_dir ${op_dir} NAME)
    set(open_dir)
    check_open_sources(${op_dir} open_dir)
    if (open_dir)
        set(open_dir ${open_dir}/op_host)
        file(GLOB OPEN_OPINFER_SRCS ${open_dir}/*_infershape*.cpp)
        file(GLOB OPEN_OPINFER_SRCS_V1 ${open_dir}/*_infer_v1.cpp)
    endif()
    file(GLOB OPINFER_SRCS ${MODULE_DIR}/*_infershape*.cpp)
    file(GLOB OPINFER_SRCS_V1 ${MODULE_DIR}/*_infer_v1.cpp)
    if (NOT OPEN_OPINFER_SRCS AND OPINFER_SRCS)
        add_infer_modules(${MODULE_HOSTNAME})
        target_sources(${MODULE_HOSTNAME}_infer_obj ${MODULE_MODE} ${OPINFER_SRCS})
    endif()
    if (NOT OPEN_OPINFER_SRCS_V1 AND OPINFER_SRCS_V1)
        add_v1_infer_modules(${MODULE_HOSTNAME})
        target_sources(${MODULE_HOSTNAME}_v1_infer_obj ${MODULE_MODE} ${OPINFER_SRCS_V1})
    endif()

    file(GLOB OPAPI_SRCS ${MODULE_DIR}/op_api/*.cpp)
    file(GLOB OPEN_OPAPI_SRCS ${open_dir}/op_api/*.cpp)
    file(GLOB OPEN_OPAPI_SRCS1 ${open_dir}/../op_api/*.cpp)
    if (NOT OPEN_OPAPI_SRCS AND NOT OPEN_OPAPI_SRCS1 AND OPAPI_SRCS)
        add_opapi_modules(${MODULE_HOSTNAME})
        target_sources(${MODULE_HOSTNAME}_opapi_obj ${MODULE_MODE} ${OPAPI_SRCS})
    endif()

    file(GLOB OPTILING_SRCS ${MODULE_DIR}/*_tiling*.cpp)
    file(GLOB OPEN_OPTILING_SRCS ${open_dir}/*_tiling*.cpp ${open_dir}/*/*_tiling*.cpp)
    message(STATUS "ENABLE_IN_OPP=${MODULE_ENABLE_IN_OPP}")
    if (NOT OPEN_OPTILING_SRCS AND OPTILING_SRCS OR MODULE_ENABLE_IN_OPP)
        add_tiling_modules(${MODULE_HOSTNAME})
        target_sources(${MODULE_HOSTNAME}_tiling_obj ${MODULE_MODE} ${OPTILING_SRCS})
    endif()

    file(GLOB_RECURSE OPTILING_SRCS ${MODULE_DIR}/op_tiling/*.cpp)
    file(GLOB_RECURSE OPEN_OPTILING_SRCS ${open_dir}/op_tiling/*.cpp)

    if (NOT OPEN_OPTILING_SRCS AND OPTILING_SRCS OR MODULE_ENABLE_IN_OPP)
        # 分别计算每个条件
        set(_is_ops_conv FALSE)
        if(MODULE_DIR MATCHES "ops-conv/common")
            set(_is_ops_conv TRUE)
        endif()

        set(_conv2d_exists FALSE)
        set(_conv2d_path "${MODULE_DIR}/../../../ops-nn/conv/conv2d_v2")
        if(EXISTS "${_conv2d_path}")
            set(_conv2d_exists TRUE)
        endif()

        message(STATUS "MODULE_DIR:${MODULE_DIR}")
        message(STATUS "_is_ops_conv:${_is_ops_conv}, _conv2d_exists:${_conv2d_exists}")
        # 组合条件判断
        if(_is_ops_conv AND _conv2d_exists)
            # 条件满足时不执行任何操作
        else()
            add_tiling_modules(${MODULE_HOSTNAME})
            target_sources(${MODULE_HOSTNAME}_tiling_obj ${MODULE_MODE} ${OPTILING_SRCS})
        endif()
    endif()

    file(GLOB_RECURSE CONV3D_V2_PRIVATE_SRCS ${MODULE_DIR}/op_tiling/conv3d_v2_*.cpp)
    file(GLOB_RECURSE CONV3D_V2_OPEN_SRCS ${open_dir}/op_tiling/conv3d_v2_*.cpp)

    if (NOT CONV3D_V2_OPEN_SRCS AND CONV3D_V2_PRIVATE_SRCS)
        target_sources(${MODULE_HOSTNAME}_tiling_obj ${MODULE_MODE} ${CONV3D_V2_PRIVATE_SRCS})
    endif()

    if (MODULE_OPTYPE)
        list(LENGTH MODULE_OPTYPE len)
        math(EXPR len "${len} - 1")
        foreach(i RANGE ${len})
            list(GET MODULE_OPTYPE ${i} op_type)
            list(GET MODULE_ACLNNTYPE ${i} aclnn_type)
            if (${aclnn_type} STREQUAL "aclnn" OR
                ${aclnn_type} STREQUAL "aclnn_inner" OR
                ${aclnn_type} STREQUAL "aclnn_exclude")
                file(GLOB OPDEF_SRCS RELATIVE ${MODULE_DIR} ${MODULE_DIR}/${op_type}_def*.cpp)
                file(GLOB OPEN_OPDEF_SRCS ${open_dir}/${op_type}_def*.cpp)
                if (NOT OPEN_OPDEF_SRCS AND OPDEF_SRCS)
                    target_sources(${MODULE_HOSTNAME}_opdef_${aclnn_type}_obj INTERFACE ${OPDEF_SRCS})
                endif()
            else()
                message(FATAL_ERROR "ACLNN TYPE UNSPPORTED, ONLY SUPPORT aclnn/aclnn_inner/aclnn_exclude")
            endif()
        endforeach()
    else()
        file(GLOB OPDEF_SRCS ${MODULE_DIR}/*_def*.cpp)
        file(GLOB OPEN_OPDEF_SRCS ${open_dir}/*_def*.cpp)
        set(L2_SRCS ${OPAPI_SRCS})
        set(L0_SRCS ${OPAPI_SRCS})
        list(FILTER L2_SRCS INCLUDE REGEX "aclnn")
        list(FILTER L0_SRCS EXCLUDE REGEX "aclnn")
        if(NOT OPEN_OPDEF_SRCS AND OPDEF_SRCS)
            if(L2_SRCS AND L0_SRCS)
                # both have L0 and L2, belong to aclnn_exclude
                target_sources(${MODULE_HOSTNAME}_opdef_aclnn_exclude_obj INTERFACE ${OPDEF_SRCS})
            elseif(NOT L2_SRCS AND L0_SRCS)
                # have l0 with no l2, belong to aclnn_exclude
                target_sources(${MODULE_HOSTNAME}_opdef_aclnn_exclude_obj INTERFACE ${OPDEF_SRCS})
            elseif(NOT L2_SRCS AND NOT L0_SRCS)
                # no L0 and L2, belong to aclnn
                target_sources(${MODULE_HOSTNAME}_opdef_aclnn_obj INTERFACE ${OPDEF_SRCS})
            else()
                message(FATAL_ERROR
                "Cannot Automatic Indentify, Should Manually specify aclnn/aclnn_inner/aclnn_exclude\n"
                "usage: add_modules_sources(HOSTNAME hostname MODE mode DIR dir OPTYPE optypes ACLNNTYPE aclnntypes)\n"
                )
            endif()
        endif()
    endif()
endfunction()

function (append_include_target op_objs type target)
    if(NOT TARGET ${target})
        message(STATUS "=== Warning<${op_objs}, add ${target} include dirs>: ops.CMakeLists.txt Target ${target} is not define")

        get_target_property(INCLUDE_DIRS ${target} INCLUDE_DIRECTORIES)

        if(INCLUDE_DIRS)
            target_include_directories(${op_objs} ${type} ${INCLUDE_DIRS})
        endif()
    endif()
endfunction()
